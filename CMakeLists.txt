cmake_minimum_required(VERSION 3.16)
project(starpath_3d_navigation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenCV REQUIRED)
set(ROS_DEPS rclcpp ament_index_cpp geometry_msgs nav_msgs sensor_msgs std_msgs builtin_interfaces visualization_msgs tf2 tf2_ros tf2_geometry_msgs cv_bridge pcl_conversions message_filters)
foreach(dep ${ROS_DEPS})
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Bspline.msg" "msg/DataDisp.msg" "msg/ChassisPath.msg" "srv/NavCommand.srv" DEPENDENCIES builtin_interfaces geometry_msgs std_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
set(V vendor)
add_library(scan345_algorithms STATIC
  ${V}/plan_env/src/grid_map.cpp ${V}/plan_env/src/raycast.cpp
  ${V}/path_searching/src/dyn_a_star.cpp
  ${V}/bspline_opt/src/uniform_bspline.cpp ${V}/bspline_opt/src/bspline_optimizer.cpp ${V}/bspline_opt/src/gradient_descent_optimizer.cpp
  ${V}/traj_utils/src/planning_visualization.cpp ${V}/traj_utils/src/polynomial_traj.cpp)
target_include_directories(scan345_algorithms PUBLIC
  ${V}/plan_env/include ${V}/path_searching/include ${V}/bspline_opt/include ${V}/traj_utils/include
  ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(scan345_algorithms ${ROS_DEPS})
target_link_libraries(scan345_algorithms Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS})
add_executable(scan345_planner ${V}/plan_manage/src/scan_planner_node.cpp ${V}/plan_manage/src/scan_replan_fsm.cpp ${V}/plan_manage/src/planner_manager.cpp)
target_include_directories(scan345_planner PRIVATE ${V}/plan_manage/include)
ament_target_dependencies(scan345_planner ${ROS_DEPS})
target_link_libraries(scan345_planner scan345_algorithms "${typesupport}")
add_executable(scan345_chassis_planner src/chassis_planner.cpp)
target_include_directories(scan345_chassis_planner PRIVATE include ${V}/plan_env/include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(scan345_chassis_planner rclcpp nav_msgs geometry_msgs sensor_msgs std_msgs pcl_conversions)
target_link_libraries(scan345_chassis_planner Eigen3::Eigen ${PCL_LIBRARIES} "${typesupport}")
if(BUILD_TESTING)
  enable_testing()
  add_executable(scan345_approach_policy_test test/approach_policy_test.cpp)
  target_include_directories(scan345_approach_policy_test PRIVATE include)
  target_compile_options(scan345_approach_policy_test PRIVATE -UNDEBUG)
  add_test(NAME scan345_approach_policy COMMAND scan345_approach_policy_test)
  add_executable(scan345_chassis_test test/chassis_search_test.cpp)
  target_include_directories(scan345_chassis_test PRIVATE include)
  add_executable(scan345_astar_test test/astar_regression.cpp)
  target_link_libraries(scan345_astar_test scan345_algorithms)
  ament_target_dependencies(scan345_astar_test rclcpp)
endif()
install(TARGETS scan345_planner scan345_chassis_planner DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/nav_nav2_runtime.py scripts/nav_global_map.py scripts/nav_runtime.py scripts/nav_ui.py scripts/nav_watchdog.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/nav_ground.py scripts/nav_motion_feedback.py scripts/nav_drive.py scripts/nav_native_limits.py scripts/nav_math.py scripts/nav_safety.py scripts/nav_view.py scripts/nav_body_model.py scripts/nav_chassis.py scripts/nav_hard_self.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/nav_link_self.py scripts/nav_envelope.py scripts/nav_static_map.py scripts/nav_grid.py scripts/nav_nav2_math.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/nav_footprint.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/nav_speckle.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/nav_obstacle_wait.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES NATIVE_NAVIGATION.md README.md NOTICE LICENSE UPSTREAM_SHA256.json DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)

foreach(dep nav2_core nav2_controller nav2_mppi_controller nav2_costmap_2d pluginlib angles)
  find_package(${dep} REQUIRED)
endforeach()
add_library(scan345_approach_mppi SHARED src/approach_mppi.cpp)
target_include_directories(scan345_approach_mppi PRIVATE include)
ament_target_dependencies(scan345_approach_mppi rclcpp nav2_core nav2_controller nav2_mppi_controller nav2_costmap_2d pluginlib angles tf2 tf2_ros tf2_geometry_msgs geometry_msgs nav_msgs std_msgs)
pluginlib_export_plugin_description_file(nav2_core approach_plugins.xml)
add_library(scan345_approach_critics SHARED src/approach_critics.cpp)
target_include_directories(scan345_approach_critics PRIVATE include)
ament_target_dependencies(scan345_approach_critics nav2_mppi_controller pluginlib)
target_link_libraries(scan345_approach_critics scan345_approach_mppi)
pluginlib_export_plugin_description_file(nav2_mppi_controller approach_critics.xml)
install(TARGETS scan345_approach_critics LIBRARY DESTINATION lib)
install(TARGETS scan345_approach_mppi LIBRARY DESTINATION lib)
install(FILES JOINT_APPROACH.md DESTINATION share/${PROJECT_NAME})

ament_package()

install(FILES scripts/nav_collision_visual.py DESTINATION lib/${PROJECT_NAME})
